% ---------------------------------------------------------------------

\newcommand{\IsoBar}[6]{%
  \begin{scope}[shift={(#1,#2)},line join=round]
    \pgfmathsetmacro{\bhw}{#4/2}%
    \fill[#6!70!nmrNavy] (\bhw,0) -- ({\bhw+#5},#5) -- ({\bhw+#5},{#3+#5}) -- (\bhw,#3) -- cycle;
    \fill[#6!45!white]  (-\bhw,#3) -- (\bhw,#3) -- ({\bhw+#5},{#3+#5}) -- ({-\bhw+#5},{#3+#5}) -- cycle;
    \fill[#6] (-\bhw,0) rectangle (\bhw,#3);
    \draw[#6!45!ink,line width=0.3pt]
      (-\bhw,0) -- (\bhw,0) -- ({\bhw+#5},#5) -- ({\bhw+#5},{#3+#5})
      -- ({-\bhw+#5},{#3+#5}) -- (-\bhw,#3) -- cycle;
    \draw[#6!45!ink,line width=0.3pt] (\bhw,0) -- (\bhw,#3) -- ({\bhw+#5},{#3+#5});
    \draw[#6!45!ink,line width=0.3pt] (-\bhw,#3) -- (\bhw,#3);
  \end{scope}%
}

% Miniature 2.5D legend swatch at {x,y} in colour {c}
\newcommand{\IsoKey}[2]{%
  \begin{scope}[shift={(#1)},line join=round]
    \fill[#2!70!nmrNavy] (1.3,0) -- (2.1,0.8) -- (2.1,2.1) -- (1.3,1.3) -- cycle;
    \fill[#2!45!white] (0,1.3) -- (1.3,1.3) -- (2.1,2.1) -- (0.8,2.1) -- cycle;
    \fill[#2] (0,0) rectangle (1.3,1.3);
    \draw[#2!45!ink,line width=0.25pt]
      (0,0) -- (1.3,0) -- (2.1,0.8) -- (2.1,2.1) -- (0.8,2.1) -- (0,1.3) -- cycle;
    \draw[#2!45!ink,line width=0.25pt] (1.3,0) -- (1.3,1.3) -- (2.1,2.1);
    \draw[#2!45!ink,line width=0.25pt] (0,1.3) -- (1.3,1.3);
  \end{scope}%
}

}
