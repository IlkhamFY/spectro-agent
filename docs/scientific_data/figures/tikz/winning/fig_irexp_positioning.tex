\newcommand{\IsoBar}[6]{%
  \begin{scope}[shift={(#1,#2)},line join=round]
    \pgfmathsetmacro{\bhw}{#4/2}%
    \fill[#6!70!nmrNavy] (\bhw,0) -- ({\bhw+#5},#5) -- ({\bhw+#5},{#3+#5}) -- (\bhw,#3) -- cycle;
    \fill[#6!45!white]  (-\bhw,#3) -- (\bhw,#3) -- ({\bhw+#5},{#3+#5}) -- ({-\bhw+#5},{#3+#5}) -- cycle;
    \fill[#6] (-\bhw,0) rectangle (\bhw,#3);
    \draw[#6!45!ink,line width=0.3pt]
      (-\bhw,0) -- (\bhw,0) -- ({\bhw+#5},#5) -- ({\bhw+#5},{#3+#5})
      -- ({-\bhw+#5},{#3+#5}) -- (-\bhw,#3) -- cycle;
    \draw[#6!45!ink,line width=0.3pt] (\bhw,0) -- (\bhw,#3) -- ({\bhw+#5},{#3+#5});
    \draw[#6!45!ink,line width=0.3pt] (-\bhw,#3) -- (\bhw,#3);
  \end{scope}%
}
